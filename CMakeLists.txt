cmake_minimum_required(VERSION 3.0.2)
project(mineguard_robot)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs nav_msgs sensor_msgs std_msgs tf)
find_package(gazebo REQUIRED)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_compile_options(-std=c++11)
add_library(mineguard_course_drive SHARED src/course_effort_drive.cpp)
add_library(mineguard_lio_lidar SHARED src/lio_lidar.cpp)
foreach(plugin mineguard_course_drive mineguard_lio_lidar)
  target_link_libraries(${plugin} ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
endforeach()
install(TARGETS mineguard_course_drive mineguard_lio_lidar LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY launch urdf meshes DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
